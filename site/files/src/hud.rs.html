           let stance = match (p.crouching, p.guarding) {                (true, true) => " (crouch, guard)",                (true, false) => " (crouch)",                (false, true) => " (guard)",                (false, false) => "",            };            format!("{gait}{stance}")        }        State::Act(a) => {            let def = a.id.def();            let charge = match def.charge {                Some((from, to)) if a.f >= from && a.f < to => "  charging",                Some((_, to)) if a.f >= to => "  CHARGED",                _ => "",            };
